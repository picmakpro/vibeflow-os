````
```sh
_GSD_ROOT="${RUNTIME_DIR:-$(git rev-parse --show-toplevel 2>/dev/null || pwd)}"
GSD_TOOLS=""
for _c in "$_GSD_ROOT/gsd-core/bin/gsd-tools.cjs" \
          "$_GSD_ROOT/.claude/gsd-core/bin/gsd-tools.cjs" \
````
