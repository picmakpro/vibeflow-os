def ecrire(chemin, texte):
    os.makedirs(os.path.dirname(chemin), exist_ok=True)
    with open(chemin, "w", encoding="utf-8") as fh:

def creuser(chemin):

def lab(nom):

def charge(l, evenement, **extra):
    objet = {"session_id": "rss", "transcript_path": "t.jsonl", "cwd": l, "hook_event_name": evenement}
    objet.update(extra)
    return json.dumps(objet, ensure_ascii=False, separators=(",", ":")).encode("utf-8")


def ecriture(l, rel):

    rss = int(m.group(1)) if m else -1

            objet = json.loads(out.split("\n")[-1])
        except ValueError:
            objet = {}

            raison = ",".join(e.get("motif", "") for e in objet.get("sans_preuve", []))
        else:
            sortie = "CTX"
    print("RSS %-18s version=%s rc=%d sortie=%s duree=%.1fs rss=%d raison=%s" % (etiquette, version, p.returncode, sortie, duree, rss,

def visiter(n, f):
    if isinstance(n, ast.FunctionDef):
        f = n.name
    if isinstance(n, ast.Call):
        fn = n.func
        nom = fn.attr if isinstance(fn, ast.Attribute) else getattr(fn, "id", None)
        base = fn.value.id if isinstance(fn, ast.Attribute) and isinstance(fn.value, ast.Name) else None
        if (base == "os" and nom in ("listdir", "scandir", "walk")) or nom in ("glob", "iglob", "iterdir", "rglob"):
            enum.append((n.lineno + debut, f, "%s.%s" % (base, nom)))
        if nom in ("read", "read_text", "read_bytes", "readline", "readlines") or (base == "json" and nom == "load"):
            args = n.args[1:] if (base == "os" and nom == "read") else n.args
            lect.append((n.lineno + debut, f, "%s.%s" % (base or "?", nom), ast.unparse(args[0]) if args else None))
    for e in ast.iter_child_nodes(n):
        visiter(e, f)


visiter(arbre, "<module>")
for e in enum:
    print("ENUM l.%d %s %s" % e)
for e in lect:
    print("LECTURE l.%d %s %s borne=%s" % e)
sans = sorted({e[1] for e in lect if e[3] is None})
fe, fl = len({e[1] for e in enum}), len({e[1] for e in lect})
print("RECENSEMENT %s : %d énumération(s) dans %d fonction(s) ; %d lecture(s) dans %d fonction(s) ; sans argument : %s" % (quand, len(enum), fe, len(lect), fl, sans))
