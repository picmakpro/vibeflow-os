par un bloc qui source `workstream-policy.sh`, définit DEUX fonctions réutilisées par la bascule
ET par la mesure réelle (une seule logique testée, jamais une resémantisée pour de vrai), puis
exécute la bascule fixture avant le checkout réel :
```yaml
      - name: check-state-integrity (anti-régression du frontmatter STATE.md par compartiment, ADR-063)
        run: |
          set -eu
          . plugin/planning-core/scripts/workstream-policy.sh

          # Classification D-02 UNCONDITIONNELLE, par lecture directe du frontmatter — jamais une
          # dépendance de forme sur un libellé que check-state-integrity.sh n'émet pas (a2, D-04).
          is_non_init() { # <fichier STATE.md> -> rc 0 = non-initialisé, rc 1 = pas ce cas
            f="$1"
            [ -r "$f" ] || return 1
            fm="$(awk '/^---[[:space:]]*$/{n++; if(n==1) next; if(n==2) exit} {print}' "$f")"
            keys="$(printf '%s\n' "$fm" | awk -F: 'NF>=2{gsub(/^[[:space:]]+|[[:space:]]+$/,"",$1); print $1}' | sort -u | awk 'NF')"
            nkeys="$(printf '%s\n' "$keys" | wc -l | tr -d ' ')"
            [ "$nkeys" -eq 2 ] || return 1
            printf '%s\n' "$keys" | grep -qx 'workstream' || return 1
            printf '%s\n' "$keys" | grep -qx 'created' || return 1
            return 0
          }

          fanout_check_state_integrity() { # <racine .planning> -> bilan sur stdout, rc 0 = 0 écart, 1 = >=1 écart
            # NOTE (INFO, juge frais) : rel="${wsdir#"$PWD"/}" dépouille par préfixe littéral, ce
            # qui peut ne pas dépouiller correctement sur un checkout SYMLINKÉ (wsdir résolu vs
            # $PWD non résolu, ou l'inverse). Risque ACCEPTÉ pour ce plan : GitHub Actions checkout
            # (actions/checkout) ne produit jamais de checkout symlinké, et c'est le seul
            # environnement où ce bloc s'exécute (étape CI, pas un usage lab générique).
            root="$1"
            fail=0; n=0
            note() { echo "::error::check-state-integrity : $1"; fail=$((fail + 1)); }
            rc=0
            ws_list="$(vf_ws_enumerate "$root")" || rc=$?
            if [ "$rc" -ne 0 ]; then
              echo "::error::vf_ws_enumerate $root a rendu rc=$rc — aucun compartiment mesurable"
              return 1
            fi
            while IFS= read -r wsdir; do
              [ -n "$wsdir" ] || continue
              n=$((n + 1))
              rel="${wsdir#"$PWD"/}"
              statefile="$rel/STATE.md"
              if is_non_init "$statefile"; then
                echo "::notice::$rel : non initialisé (D-02, frontmatter réduit à workstream:+created:), jamais compté comme écart (D-06)"
                continue
              fi
              rc=0
              out="$(bash plugin/conductor/scripts/check-state-integrity.sh --file "$statefile" 2>&1)" || rc=$?
              echo "== check-state-integrity $rel : rc=$rc =="; printf '%s\n' "$out"
              [ "$rc" -eq 0 ] || note "$rel : rc=$rc, attendu 0 (conforme) ou non-initialisé détecté en amont par is_non_init"
            done <<< "$ws_list"
            [ "$n" -ge 1 ] || { echo "::error::aucun compartiment énuméré — vérité vide"; return 1; }
            echo "== BILAN check-state-integrity ($root) : $fail écart(s) sur $n compartiment(s) =="
            [ "$fail" -eq 0 ]
          }

          # --- 1/3 BASCULE sur fixture jetable, AVANT le checkout réel (QUAL-01, patron
          # check-instruction-budget) : un compartiment CORROMPU doit faire échouer le fan-out -----
          T=$(mktemp -d); trap 'rm -rf "$T"' EXIT
          git -C "$T" init -q
          mkdir -p "$T/.planning/workstreams/sain"
          cat > "$T/.planning/workstreams/sain/STATE.md" <<'EOF'
---
gsd_state_version: "1.0"
milestone: m1
current_phase: "1"
status: executing
---
Phase: 1
EOF
          mkdir -p "$T/.planning/workstreams/casse"
          cat > "$T/.planning/workstreams/casse/STATE.md" <<'EOF'
---
workstream: casse
created: "2026-09-23"
milestone: ""
---
Phase: 1
EOF
          git -C "$T" add -A
          git -C "$T" -c user.email=t@t -c user.name=t commit -q -m "fixture-casse-initiale"

          rc=0; fanout_check_state_integrity "$T/.planning" >"$T/out1" 2>&1 || rc=$?
          cat "$T/out1"
          echo "== 1/3 fixture (sain + casse) : rc=$rc =="
          [ "$rc" -eq 1 ] || { echo "::error::1/3 : rc=$rc, attendu 1 — le compartiment casse (milestone absent, 3 clés, ni nominal ni sain) doit faire échouer le fan-out"; exit 1; }
          grep -q 'casse' "$T/out1" || { echo "::error::1/3 : le compartiment fautif « casse » n'est pas nommé dans le verdict"; exit 1; }

          # --- 2/3 BASCULE INVERSE : casse réparé (frontmatter complet) -> redevient vert ----------
          cat > "$T/.planning/workstreams/casse/STATE.md" <<'EOF'
---
gsd_state_version: "1.0"
milestone: m1
current_phase: "1"
status: executing
---
Phase: 1
EOF
          rc=0; fanout_check_state_integrity "$T/.planning" >"$T/out2" 2>&1 || rc=$?
          cat "$T/out2"
          echo "== 2/3 fixture réparée : rc=$rc =="
          [ "$rc" -eq 0 ] || { echo "::error::2/3 : rc=$rc, attendu 0 — réparé, casse doit redevenir conforme"; exit 1; }

          # --- 3/3 mesure du CHECKOUT RÉEL (fiabilite conforme, gouvernance non-initialisé — ne
          # doit JAMAIS échouer à cause de gouvernance, D-06/Success Criterion 5) ------------------
          rc=0; fanout_check_state_integrity ".planning" || rc=$?
          echo "== 3/3 checkout réel : rc=$rc =="
          if [ "$rc" -ne 0 ]; then exit 1; fi
```
