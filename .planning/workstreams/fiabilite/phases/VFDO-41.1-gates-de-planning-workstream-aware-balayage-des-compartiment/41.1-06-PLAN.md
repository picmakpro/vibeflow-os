          # ROADMAP/REQUIREMENTS de contenu » est RETIRÉE — elle classait `gouvernance` CORROMPU

          # le mauvais nom. Ce qu'elle laissait passer EN SILENCE, c'est le cas INVERSE — un

            local _wsc_d="$1"
            if [ -s "$_wsc_d/ROADMAP.md" ]; then return 0; fi

          }
          # NOTE (INFO, juge frais, accepté) : un compartiment SANS STATE.md du tout (`[ -r "$f" ]`
          # faux) n'est PAS classé non-init — il tombe au script, qui rougira. C'EST VOULU : l'absence
          # totale de STATE.md n'est ni la facture nominale de `workstream create` (qui écrit TOUJOURS
          # un STATE.md) ni un état sain — c'est corrompu par construction, et le traiter en échec
          # audible (plutôt qu'en silence) est la posture fail-closed cohérente avec tout ce plan.

          # CORRECTION post plan-checker (BLOCK-A, BLOCK-B, 2e passe) : <lab_root> remplace
          # <racine .planning> — check-state-integrity.sh attend --path=RACINE DU LAB (parent de
          # .planning), --file RELATIF à cette racine (JAMAIS un chemin absolu, JAMAIS --file seul

            local _fsi_fail=0 _fsi_n=0 _fsi_rc=0 _fsi_wsdir _fsi_rel _fsi_statefile_rel _fsi_out _fsi_rc2 _fsi_ws_list

                return 1

DEUX lectures sont possibles et AUCUNE n'est choisie ici : (i) compartiment NEUF alimenté par carve-out ; (ii) compartiment TRAVAILLÉ dont le STATE.md a été TRONQUÉ en plein cycle d'écriture. Les deux ont la même signature sur disque. Ceci est une NOTICE, PAS un échec — à lever à la main."

          gsd_state_version: 1.0
          milestone: m1
          current_phase: 1
          status: executing

            total_phases: 1
            completed_phases: 0
            total_plans: 1
            completed_plans: 0

          printf '%s\n' '---' 'workstream: carve' 'created: 2026-09-23' '---' > "$T7/.planning/workstreams/carve/STATE.md"
          printf '%s\n' '---' 'workstream: vierge' 'created: 2026-09-23' '---' > "$T7/.planning/workstreams/vierge/STATE.md"
          printf '# Feuille de route\n\n## Phase 1\n' > "$T7/.planning/workstreams/carve/ROADMAP.md"

          cat "$T7/out6"
          echo "== 6/6 ambiguïté D-02 : rc=$rc =="
          [ "$rc" -eq 0 ] || { echo "::error::6/6 : rc=$rc, attendu 0 — ni le non-initialisé simple ni l'ambiguïté ne sont des échecs (D-02 amendée)"; exit 1; }
          n_amb=$(awk 'index($0, "AMBIGUITE-D02") > 0 {c++} END{print c+0}' "$T7/out6")
          [ "$n_amb" -eq 1 ] || { echo "::error::6/6 : $n_amb notice(s) AMBIGUITE-D02, attendu EXACTEMENT 1 — « carve » (contenu à côté) doit la produire, « vierge » (rien à côté) doit NE PAS la produire"; exit 1; }
          grep -q 'carve : AMBIGUITE-D02' "$T7/out6" || { echo "::error::6/6 : la notice ne nomme pas le bon compartiment"; exit 1; }
          rm -rf "$T7"
